cmake_minimum_required(VERSION 3.10)
project(autoocr_batch_ocr LANGUAGES CXX)
enable_testing()
add_subdirectory(ipc)
add_subdirectory(scheduler)
add_executable(test_jpeg_color test_jpeg_color.cpp)
target_compile_features(test_jpeg_color PRIVATE cxx_std_14)
target_compile_options(test_jpeg_color PRIVATE -UNDEBUG)
add_test(NAME jpeg_colour_rounding COMMAND test_jpeg_color)
option(AUTOOCR_BUILD_JPEG "Build the matching Jetson hardware JPEG probe" OFF)
if(NOT AUTOOCR_BUILD_JPEG)
    add_subdirectory(app)
    return()
endif()

set(JETSON_MMAPI "/usr/src/jetson_multimedia_api" CACHE PATH "Matching L4T Multimedia API source")
if(NOT EXISTS "${JETSON_MMAPI}/include/NvJpegDecoder.h")
    message(FATAL_ERROR "Set JETSON_MMAPI to the matching, verified SDK source directory")
endif()
set(COMMON "${JETSON_MMAPI}/samples/common/classes")
file(SHA256 "${COMMON}/NvJpegDecoder.cpp" SDK_JPEG_SHA)
if(NOT SDK_JPEG_SHA STREQUAL "9d5e9b85a5c882e834fcdad6bf89a4314f050cb820493c7ae8d0a692f37814ce")
    message(FATAL_ERROR "Expected the verified L4T32.7.1 NvJpegDecoder source")
endif()
find_program(PATCH_EXECUTABLE patch)
if(NOT PATCH_EXECUTABLE)
    message(FATAL_ERROR "GNU patch is required for the project-local partial-MCU fix")
endif()
set(SAFE_JPEG "${CMAKE_CURRENT_BINARY_DIR}/NvJpegDecoderSafe.cpp")
configure_file("${COMMON}/NvJpegDecoder.cpp" "${SAFE_JPEG}" COPYONLY)
execute_process(COMMAND "${PATCH_EXECUTABLE}" --batch --fuzz=0 --no-backup-if-mismatch
    --input "${CMAKE_CURRENT_SOURCE_DIR}/patches/nvjpeg-partial-mcu.patch" "${SAFE_JPEG}"
    RESULT_VARIABLE PATCH_RESULT)
if(NOT PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "Matching SDK partial-MCU patch did not apply; do not use unpatched hardware path")
endif()
file(SHA256 "${SAFE_JPEG}" SAFE_JPEG_SHA)
if(NOT SAFE_JPEG_SHA STREQUAL "54d8d920f77193e3288b171d3e1d3aa13042054ae813ac05efcf9bac283639d2")
    message(FATAL_ERROR "Unexpected patched JPEG implementation")
endif()
set(JPEG_SOURCES
    "${SAFE_JPEG}" "${COMMON}/NvBuffer.cpp"
    "${COMMON}/NvElement.cpp" "${COMMON}/NvElementProfiler.cpp" "${COMMON}/NvLogging.cpp")
add_executable(jpeg_decode_probe jpeg_decode_probe.cpp ${JPEG_SOURCES})
target_compile_features(jpeg_decode_probe PRIVATE cxx_std_14)
target_compile_definitions(jpeg_decode_probe PRIVATE TEGRA_ACCELERATE)
target_include_directories(jpeg_decode_probe PRIVATE
    "${JETSON_MMAPI}/include" "${JETSON_MMAPI}/include/libjpeg-8b")
find_library(NVJPEG nvjpeg PATHS /usr/lib/aarch64-linux-gnu/tegra NO_DEFAULT_PATH)
if(NOT NVJPEG)
    message(FATAL_ERROR "Jetson accelerated libnvjpeg, not desktop libjpeg, is required")
endif()
find_path(V4L2_INCLUDE_DIR libv4l2.h)
find_library(V4L2_LIBRARY NAMES v4l2 libv4l2.so.0)
if(NOT V4L2_INCLUDE_DIR OR NOT V4L2_LIBRARY)
    message(FATAL_ERROR "Matching V4L2 headers and the existing runtime library are required")
endif()
target_include_directories(jpeg_decode_probe PRIVATE "${V4L2_INCLUDE_DIR}")
target_link_libraries(jpeg_decode_probe PRIVATE "${NVJPEG}" "${V4L2_LIBRARY}" pthread)
set_target_properties(jpeg_decode_probe PROPERTIES BUILD_RPATH "/usr/lib/aarch64-linux-gnu/tegra")
add_subdirectory(app)
