cmake_minimum_required(VERSION 3.10)
project(autoocr_batch_app LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(autoocr_batch_app batch_app.cpp)
target_compile_features(autoocr_batch_app PRIVATE cxx_std_14)
target_compile_options(autoocr_batch_app PRIVATE -Wall -Wextra -Werror)
target_link_libraries(autoocr_batch_app PRIVATE Threads::Threads)
add_test(NAME batch_policy_cpp COMMAND autoocr_batch_app --self-test)
if(AUTOOCR_BUILD_JPEG)
    target_sources(autoocr_batch_app PRIVATE ${JPEG_SOURCES})
    target_compile_definitions(autoocr_batch_app PRIVATE AUTOOCR_JPEG TEGRA_ACCELERATE)
    target_include_directories(autoocr_batch_app PRIVATE "${JETSON_MMAPI}/include" "${JETSON_MMAPI}/include/libjpeg-8b" "${V4L2_INCLUDE_DIR}")
    target_link_libraries(autoocr_batch_app PRIVATE "${NVJPEG}" "${V4L2_LIBRARY}" pthread)
    set_target_properties(autoocr_batch_app PROPERTIES BUILD_RPATH "/usr/lib/aarch64-linux-gnu/tegra")
endif()
