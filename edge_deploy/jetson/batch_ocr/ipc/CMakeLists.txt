cmake_minimum_required(VERSION 3.10)
project(autoocr_ipc_probe LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(ipc_probe ipc_probe.cpp)
target_compile_features(ipc_probe PRIVATE cxx_std_14)
target_compile_options(ipc_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ipc_probe PRIVATE Threads::Threads)
enable_testing()
add_test(NAME ipc_transport_cpp COMMAND ipc_probe --self-test)
add_executable(stage_client_probe stage_client_probe.cpp)
target_compile_features(stage_client_probe PRIVATE cxx_std_14)
target_compile_options(stage_client_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(stage_client_probe PRIVATE Threads::Threads)
add_test(NAME ipc_stage_cpp COMMAND stage_client_probe --self-test)
